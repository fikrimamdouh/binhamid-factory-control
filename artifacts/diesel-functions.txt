function opsFuelStats(vehicleId){
  const L=OPS.fuel.filter(x=>!vehicleId||x.vehicleId===vehicleId).slice().sort((a,b)=>String(a.date).localeCompare(String(b.date)));
  let km=0,liters=0,cost=0,eff=null;
  L.forEach(x=>{liters+=Number(x.liters||0);cost+=Number(x.cost||0);});
  if(L.length>1){km=Math.max(0,Number(L.at(-1).odometer||0)-Number(L[0].odometer||0));if(liters>0)eff=km/liters;}
  return {liters,cost,km,eff};
}

function opsFuelExcludedIdentity(vehicleName,driverName){
  const vehicleOk=opsNormArabic(vehicleName).includes(opsNormArabic(OPS_INTERNAL_FUEL_CONTROL.excludedVehicle));
  const driverOk=opsNormArabic(driverName).includes(opsNormArabic(OPS_INTERNAL_FUEL_CONTROL.excludedDriver));

function opsFuelIsoDateTime(value){
  if(value instanceof Date&&!Number.isNaN(value.getTime()))return value.toISOString().slice(0,19);
  if(typeof value==='number'&&typeof XLSX!=='undefined'&&XLSX.SSF?.parse_date_code){const d=XLSX.SSF.parse_date_code(value);if(d)return `${d.y}-${String(d.m).padStart(2,'0')}-${String(d.d).padStart(2,'0')}T${String(d.H||0).padStart(2,'0')}:${String(d.M||0).padStart(2,'0')}:${String(d.S||0).padStart(2,'0')}`;}
  const s=String(value||'').trim(),iso=s.match(/^(20\d{2})[-\/]([01]?\d)[-\/]([0-3]?\d)(?:[ T]([0-2]?\d):([0-5]?\d)(?::([0-5]?\d))?)?/);if(iso)return `${iso[1]}-${String(iso[2]).padStart(2,'0')}-${String(iso[3]).padStart(2,'0')}T${String(iso[4]||0).padStart(2,'0')}:${String(iso[5]||0).padStart(2,'0')}:${String(iso[6]||0).padStart(2,'0')}`;
  const dmy=s.match(/^([0-3]?\d)[-\/]([01]?\d)[-\/](20\d{2})(?:[ T]([0-2]?\d):([0-5]?\d)(?::([0-5]?\d))?)?/);if(dmy)return `${dmy[3]}-${String(dmy[2]).padStart(2,'0')}-${String(dmy[1]).padStart(2,'0')}T${String(dmy[4]||0).padStart(2,'0')}:${String(dmy[5]||0).padStart(2,'0')}:${String(dmy[6]||0).padStart(2,'0')}`;return '';
}

=== opsFuelMasterPlan ===
function opsFuelMasterPlan(rows){
  const vehicles=new Map(),drivers=new Map();
  (rows||[]).forEach(x=>{
    const plate=String(x.plate||'').trim(),plateKey=opsFuelPlateKey(plate),driver=String(x.driver||'').trim(),driverKey=opsNormArabic(driver),match=x.match||opsMatchVehicleByPlate(plate);
    if(plateKey&&!vehicles.has(plateKey))vehicles.set(plateKey,{plateKey,plate,vehicleName:String(x.vehicleName||'').trim(),driver,match});
    if(driverKey&&!drivers.has(driverKey))drivers.set(driverKey,{driverKey,name:driver,employee:opsFuelEmployeeByName(driver)});
  });
  const vehicleRows=[...vehicles.values()],driverRows=[...drivers.values()];
  return {vehicles:vehicleRows,drivers:driverRows,matchedVehicles:vehicleRows.filter(x=>x.match?.vehicle),missingVehicles:vehicleRows.filter(x=>!x.match?.vehicle),matchedDrivers:driverRows.filter(x=>x.employee),missingDrivers:driverRows.filter(x=>!x.employee)};
}

  const plan=opsFuelMasterPlan(rows),employeeMap=new Map(),vehicleMap=new Map();
  plan.drivers.forEach(x=>{if(x.employee)employeeMap.set(x.driverKey,x.employee);});
  plan.vehicles.forEach(x=>{const vehicle=x.match?.vehicle||opsMatchVehicleByPlate(x.plate).vehicle;if(vehicle)vehicleMap.set(x.plateKey,vehicle);});
  return {plan,employeeMap,vehicleMap,addedEmployees:[],addedVehicles:[]};
}

function opsDieselConsecutiveRuns(rows){
  const groups=new Map(),dayNumber=s=>{const [y,m,d]=String(s||'').slice(0,10).split('-').map(Number);return y&&m&&d?Math.floor(Date.UTC(y,m-1,d)/86400000):NaN;};
  (rows||[]).forEach(f=>{const key=opsFuelGroupKey(f),date=String(f.date||'').slice(0,10);if(!date)return;if(!groups.has(key))groups.set(key,{key,plate:opsFuelDisplayPlate(f),vehicle:opsFuelDisplayVehicle(f),driver:opsFuelDisplayDriver(f),days:new Map()});const g=groups.get(key),x=g.days.get(date)||{date,fills:0,liters:0};x.fills++;x.liters+=Number(f.liters||0);g.days.set(date,x);});
  const runs=[];
  groups.forEach(g=>{const days=[...g.days.values()].sort((a,b)=>a.date.localeCompare(b.date));let run=[];const flush=()=>{if(run.length>=2)runs.push({key:g.key,plate:g.plate,vehicle:g.vehicle,driver:g.driver,dates:run.map(x=>x.date),days:run.length,fills:run.reduce((s,x)=>s+x.fills,0),liters:run.reduce((s,x)=>s+x.liters,0),from:run[0].date,to:run.at(-1).date});run=[];};days.forEach(x=>{if(!run.length||dayNumber(x.date)-dayNumber(run.at(-1).date)===1)run.push(x);else{flush();run=[x];}});flush();});
  return runs.sort((a,b)=>b.days-a.days||a.plate.localeCompare(b.plate,'ar'));
}

function opsDieselControlData(rows){
  const priceVariance=OPS_INTERNAL_FUEL_CONTROL.priceVariancePct,volumeVariance=OPS_INTERNAL_FUEL_CONTROL.volumeVariancePct,rapidHours=OPS_INTERNAL_FUEL_CONTROL.rapidRefillHours;
  const ordered=rows.slice().sort((a,b)=>String(a.date).localeCompare(String(b.date))),prices=ordered.map(opsFuelUnitPrice).filter(x=>x>0),medianPrice=opsMedian(prices),byVehicleLiters=new Map(),receiptMap=new Map(),alerts=[];
  ordered.forEach(f=>{const key=opsFuelGroupKey(f);if(!byVehicleLiters.has(key))byVehicleLiters.set(key,[]);byVehicleLiters.get(key).push(Number(f.liters||0));const receipt=opsNormArabic(f.receipt||'');if(receipt){if(!receiptMap.has(receipt))receiptMap.set(receipt,[]);receiptMap.get(receipt).push(f);}});
  const vehicleMedians=new Map([...byVehicleLiters].map(([k,v])=>[k,opsMedian(v)])),previousByVehicle=new Map(),add=(f,level,check,detail)=>alerts.push({fuelId:f.id,level,check,detail,date:f.date,plate:opsFuelDisplayPlate(f),receipt:f.receipt||f.no||'—'});
  ordered.forEach(f=>{
    const key=opsFuelGroupKey(f),liters=Number(f.liters||0),cost=Number(f.cost||0),price=opsFuelUnitPrice(f),current=Number(f.odometer||0),previousOdo=Number(f.previousOdometer||0),receipt=opsNormArabic(f.receipt||''),previous=previousByVehicle.get(key);
    if(receipt&&(receiptMap.get(receipt)?.length||0)>1)add(f,'danger','إيصال مكرر',`رقم الإيصال مستخدم في ${receiptMap.get(receipt).length} تعبئات`);
    if(!receipt)add(f,'warn','إيصال غير مسجل','لا يوجد رقم إيصال يمكن المطابقة عليه');
    if(!String(f.station||'').trim())add(f,'info','المحطة غير مسجلة','استكمال اسم محطة الوقود');
    if(!String(opsFuelDisplayDriver(f)).trim())add(f,'info','السائق غير مسجل','استكمال اسم السائق أو ربطه بالموظفين');
    if(current>0&&previousOdo>0&&current<=previousOdo)add(f,'danger','قراءة عداد غير منطقية',`السابق ${opsNum(previousOdo)} والحالي ${opsNum(current)}`);
    else if(current<=0)add(f,'info','العداد غير مسجل','لا يمكن حساب كفاءة الاستهلاك لهذه التعبئة');
    if(previous){const hours=(new Date(f.date).getTime()-new Date(previous.date).getTime())/3600000;if(Number.isFinite(hours)&&hours>=0&&hours<=rapidHours)add(f,'warn','تعبئة متقاربة',`الفاصل عن التعبئة السابقة ${hours.toFixed(1)} ساعة`);}
    const medianLiters=vehicleMedians.get(key)||0,vehicleCount=byVehicleLiters.get(key)?.length||0;if(vehicleCount>=3&&medianLiters>0&&liters>medianLiters*(1+volumeVariance/100))add(f,'warn','كمية أعلى من المعتاد',`${opsNum(liters)} لتر مقابل وسيط ${opsNum(medianLiters)} لتر للمركبة`);
    if(prices.length>=3&&medianPrice>0&&price>0&&Math.abs(price-medianPrice)/medianPrice*100>priceVariance)add(f,'warn','سعر لتر مختلف',`${opsMoney(price)} ر.س مقابل وسيط ${opsMoney(medianPrice)} ر.س`);
    if(liters<=0||cost<=0)add(f,'danger','قيمة تعبئة غير صالحة',`اللترات ${opsNum(liters)} والتكلفة ${opsMoney(cost)}`);
    previousByVehicle.set(key,f);
  });
  const rank={danger:0,warn:1,info:2};alerts.sort((a,b)=>(rank[a.level]-rank[b.level])||String(a.date).localeCompare(String(b.date)));
  const byFuel=new Map();alerts.forEach(a=>{if(!byFuel.has(a.fuelId))byFuel.set(a.fuelId,[]);byFuel.get(a.fuelId).push(a);});
  return {alerts,byFuel,consecutiveRuns:opsDieselConsecutiveRuns(rows),medianPrice,priceVariance,volumeVariance,rapidHours,danger:alerts.filter(x=>x.level==='danger').length,warn:alerts.filter(x=>x.level==='warn').length,info:alerts.filter(x=>x.level==='info').length};
}

function opsDieselVehicleSummary(rows,control){

  rows.forEach(f=>{const key=opsFuelGroupKey(f),driver=opsFuelDisplayDriver(f),r=map.get(key)||{key,vehicleId:f.vehicleId,plate:opsFuelDisplayPlate(f),vehicle:`${opsFuelDisplayVehicle(f)}${driver&&driver!=='—'?' — السائق: '+driver:''}`,driver,fills:0,liters:0,amount:0,distance:0,alerts:0};r.fills++;r.liters+=Number(f.liters||0);r.amount+=Number(f.cost||0);r.distance+=opsFuelDistance(f);r.alerts+=(control?.byFuel.get(f.id)?.length||0);map.set(key,r);});
  return [...map.values()].map(r=>({...r,avgFill:r.fills?r.liters/r.fills:0,avgPrice:r.liters?r.amount/r.liters:0,kmPerLiter:r.liters&&r.distance?r.distance/r.liters:0,costPerKm:r.distance?r.amount/r.distance:0,share:totalLiters?r.liters/totalLiters*100:0})).sort((a,b)=>b.liters-a.liters);
